/// Chapter heading — Exhibit I
/// Generates: CHAPTER N [triple space] TITLE IN ALL CAPS
/// Spacing between lines of title (AU DMin Manual p. 13 & Exhibit I):
/// - 2 lines: double spacing (word-double-leading)
/// - 3+ lines: single spacing (word-single-leading)
#let dmin-chapter(title: "", lines: auto) = [
  #let num-lines = if lines != auto { lines } else { count-title-lines(title) }
  #let effective-lines = if num-lines > 1 { num-lines } else { auto }
  #dmin-chapter-lines.update(effective-lines)
  #let formatted = format-title(title)
  #heading(level: 1, numbering: "1.", outlined: true)[#formatted]
  #dmin-chapter-lines.update(auto)
  #consecutive-subhead.update(true)
]

/// Level 5 Subheading (AU Level 5): Run-in to paragraph
/// Indented 0.5", Bold, sentence style, ends with period (Ch.2 p.10)
/// Preceded by triple space (above: 28pt). Followed by double space (built into paragraph).
/// Usage: #subhead5[Students feel responsible][When the students...]
#let subhead5(title, body) = context {
  let was-subhead = consecutive-subhead.get()
  let abv = if was-subhead { 0pt } else { word-space-skip-2 }
  consecutive-subhead.update(false)
  let formatted-title = if type(title) == str {
    let t = title.trim()
    if t.ends-with(".") { t } else { t + "." }
  } else {
    let r = repr(title)
    if r.ends-with(".]") or r.ends-with(".\"") or r.ends-with(".") {
      title
    } else {
      [#title.]
    }
  }
  v(abv, weak: true)
  text(weight: "bold")[#formatted-title]
  [ ]
  body
}

/// Block Quote — Chapter 3 (p. 12)
/// 5+ lines of quoted material; single-spaced; indented 0.25in (quarter inch)
/// Double space before and after (above: 14pt, below: 14pt)
#let blockquote(body) = [
  #block(width: 100%, above: word-space-skip-1, below: word-space-skip-1)[
    #pad(left: 0.25in)[
      #set par(leading: word-single-leading, spacing: word-single-leading, first-line-indent: 0pt, justify: false)
      #set text(size: 12pt)
      #body
    ]
  ]
]

/// DMin Table — Chapter 2 (p. 7-8) & Chapter 3 (p. 13) & Exhibit F
/// Quadruple space (skip 3 lines = 42pt) before/after when on page with text
#let dmin-table(number: auto, title: "", table-content, note: none, breakable: false) = context {
  let s = dmin-get-strings(au-dmin-lang.get())
  let formatted = format-title(title)
  let content = [
    #table-content
    #if note != none [
      #v(6pt, weak: true)
      #set text(size: 10pt)
      #set par(leading: word-single-leading, first-line-indent: 0pt)
      #note
    ]
  ]
  figure(
    content,
    caption: figure.caption(position: top)[#formatted],
    kind: table,
    supplement: s.table,
  )
}

/// DMin Isolated Table — Chapter 2 (p. 7, Rule 3) & Exhibit F
/// Placed on its own separate page (pagebreak before and after; no body text sharing the page)
#let dmin-table-isolated(number: auto, title: "", table-content, note: none, breakable: false) = [
  #pagebreak(weak: true)
  #dmin-table(number: number, title: title, table-content, note: note, breakable: breakable)
  #pagebreak()
]

/// DMin Figure — Chapter 2 (p. 7-8) & Chapter 3 (p. 13) & Exhibit F
/// Quadruple space (skip 3 lines = 42pt) before/after when on page with text
#let dmin-figure(number: auto, caption: "", figure-content, source: none, breakable: false) = context {
  let s = dmin-get-strings(au-dmin-lang.get())
  let formatted = format-title(caption)
  let content = [
    #figure-content
    #if source != none [
      #v(6pt, weak: true)
      #set text(size: 10pt, weight: "regular")
      #set par(leading: word-single-leading, first-line-indent: 0pt)
      #source
    ]
  ]
  figure(
    content,
    caption: figure.caption(position: bottom)[#formatted],
    kind: image,
    supplement: s.figure,
  )
}

/// DMin Isolated Figure — Chapter 2 (p. 7, Rule 3) & Exhibit F
/// Placed on its own separate page (pagebreak before and after; no body text sharing the page)
#let dmin-figure-isolated(number: auto, caption: "", figure-content, source: none) = [
  #pagebreak(weak: true)
  #dmin-figure(number: number, caption: caption, figure-content, source: source)
  #pagebreak()
]

/// Appendix Section — Chapter 2 (p. 11)
/// If cover-sheet: true, creates a centered divider sheet (APPENDIX A + TITLE) and puts body on next page without repeating heading.
/// If cover-sheet: false (default), starts page with APPENDIX A + TITLE at top (2" margin) and body below.
#let appendix-section(title: "", cover-sheet: false, body) = [
  #let formatted = format-title(title)
  #appendix-counter.step()
  #if cover-sheet [
    #pagebreak(weak: true)
    #set page(footer: none)
    #v(2.5in)
    #align(center)[
      #set text(size: 12pt)
      #context {
        let s = dmin-get-strings(au-dmin-lang.get())
        [#s.appendix #appendix-counter.display("A") \ ]
      }
      #v(word-space-skip-2)
      #context {
        let test-m = measure(block(width: 100%)[
          #set text(size: 12pt)
          #set par(leading: 0pt, first-line-indent: 0pt)
          #upper(formatted)
        ])
        let title-leading = if test-m.height >= 35pt { word-single-leading } else { word-double-leading }
        block(width: 100%)[
          #set par(leading: title-leading, first-line-indent: 0pt)
          #upper(formatted)
        ]
      }
    ]
    #heading(level: 1, numbering: "A.", supplement: [Appendix-cover], outlined: true)[#formatted]
    #pagebreak()
    #set par(leading: word-double-leading, spacing: word-double-leading, first-line-indent: 0.5in, justify: false)
    #body
  ] else [
    #heading(level: 1, numbering: "A.", supplement: [Appendix], outlined: true)[#formatted]
    #set par(leading: word-double-leading, spacing: word-double-leading, first-line-indent: 0.5in, justify: false)
    #body
  ]
]

/// Reference List — Chapter 5 & Appendix A
#let dmin-bibliography(bib-file, style: "../csl/andrews-university-dmin-author-date.csl", title: auto, full: false) = [
  #context {
    let s = dmin-get-strings(au-dmin-lang.get())
    let display-title = if title == auto { s.ref_list } else { upper(title) }
    heading(level: 1, numbering: none, outlined: true)[#display-title]
  }

  // Забороняє розривати будь-яке окреме джерело навпіл між сторінками
  #show bibliography: set grid.cell(breakable: false)

  #set par(
    leading: word-single-leading,
    spacing: word-space-skip-2,
    hanging-indent: 0.5in,
    first-line-indent: 0pt,
    justify: false,
  )
  #bibliography(bib-file, style: style, title: none, full: full)
]

// ==============================================================================
// 📐 AU DMin Layout, Typography, Spacing & Show Rules
// Conforms to: Doctor of Ministry Project Writing Manual (4th Edition, Aug 2019)
// ==============================================================================

#import "state.typ": *
#import "i18n.typ": *

// ==============================================================================
// 📏 Spacing Constants (Strictly integer pt values / MS Word & AU DMin Standard)
// ==============================================================================
#let word-single-leading = 2pt      // Одинарний інтервал (14pt baseline)
#let word-double-leading = 16pt     // Подвійний інтервал (28pt baseline)
#let word-space-skip-1 = 2pt       // Single space (пропуск 1 одинарного рядка)
#let word-space-skip-2 = 16pt       // Double space (пропуск 2 одинарних рядків)
#let word-space-skip-3 = 24pt       // Triple spaces (пропуск 3 одинарних рядків)
#let word-space-skip-4 = 32pt       // Four spaces (пропуск 3 одинарних рядків)


/// Helper to handle string/content line breaks with \ or \n and uppercase conversion
#let format-title(t, to-upper: false) = {
  if type(t) == str {
    let s = if to-upper { upper(t) } else { t }
    let s-norm = s.replace("\\", "\n")
    let lines = s-norm.split("\n").map(l => l.trim()).filter(l => l.len() > 0)
    if lines.len() > 1 {
      lines.join(linebreak())
    } else {
      s
    }
  } else if type(t) == content {
    if to-upper { upper(t) } else { t }
  } else {
    t
  }
}

/// Helper to determine the number of lines in a title
#let count-title-lines(t) = {
  if type(t) == str {
    let s-norm = t.replace("\\", "\n")
    let lines = s-norm.split("\n").map(l => l.trim()).filter(l => l.len() > 0)
    calc.max(1, lines.len())
  } else if type(t) == content {
    let rep = repr(t)
    let breaks = rep.matches("linebreak").len()
    breaks + 1
  } else {
    1
  }
}


#let apply-dmin-layout(lang: "en", body) = {
  // ============================================================================
  // PAGE GEOMETRY — Chapter 3 (p. 12)
  // Left margin: 1.5" (for binding); Right/Top/Bottom: 1.0"
  // ============================================================================
  set page(
    paper: "us-letter",
    margin: (
      left: 1.5in,
      right: 1.0in,
      top: 1.0in,
      bottom: 1.0in,
    ),
    header: none,
    footer: context {
      align(center)[
        #text(size: 12pt)[
          #counter(page).display()
        ]
      ]
    },
  )

  // ============================================================================
  // TYPOGRAPHY — Chapter 1 (p. 1)
  // "Fonts similar to Times New Roman... serif... 10-12 points"
  // ============================================================================
  set text(
    font: ("Times New Roman", "Liberation Serif", "Nimbus Roman"),
    size: 12pt,
    lang: lang,
    top-edge: 0.8em,
    bottom-edge: -0.2em,
    costs: (widow: 300%, orphan: 300%, runt: 300%),
    hyphenate: false,
  )

  // Запобігання висячим коротким прийменникам, сполучникам та числам з крапкою
  show regex("(?:\b|^)([а-яА-ЯіІїЇєЄґҐa-zA-Z]{1,2}|[0-9]+\.)\s"): it => {
    it.text.trim() + sym.space.nobreak
  }

  // ============================================================================
  // SPACING — Chapter 3 (p. 12-13)
  // All text: double-spaced (28pt baseline = 14pt font box + 14pt leading)
  // Ragged right margin (NOT justified)
  // Paragraph indent: 0.5 inches
  // ============================================================================
  set par(
    leading: word-double-leading,
    spacing: word-double-leading,
    first-line-indent: (amount: 0.5in, all: true),
    justify: false,
  )

  show par: it => consecutive-subhead.update(false) + it

  // ============================================================================
  // BIBLIOGRAPHY — Chapter 5 & Appendix A
  // Запобігання розриву окремих бібліографічних записів між сторінками
  // ============================================================================
  show bibliography: set grid.cell(breakable: false)

  // ============================================================================
  // FOOTNOTES — Chapter 5 (p. 21)
  // 10pt font, single-spaced within, double space between, 0.5in indent
  // ============================================================================
  show footnote.entry: it => {
    set text(size: 10pt)
    set par(leading: 2pt, spacing: 12pt, first-line-indent: 0.5in, justify: false)
    it
  }

  // ============================================================================
  // HEADINGS — Chapter 2 (p. 9-10) & Chapter 3 (p. 13-14) & Exhibit I
  // ============================================================================
  set heading(numbering: none)
  show heading: it => {
    if it.level == 1 {
      consecutive-subhead.update(true)
      if it.has("supplement") and it.supplement == [Appendix-cover] {
        // Handled by appendix cover sheet, do not render duplicate heading on content page
        none
      } else if it.has("supplement") and (it.supplement == [same-page] or repr(it.supplement) == "[same-page]") {
        // Section on same page without pagebreak (e.g., List of Tables right after List of Illustrations)
        v(word-space-skip-2)
        block(width: 100%, below: word-space-skip-1, sticky: true)[
          #align(center)[
            #set text(size: 12pt, weight: "regular")
            #set par(leading: word-single-leading, justify: false, first-line-indent: 0pt)
            #upper(it.body)
          ]
        ]
      } else {
        // Major Section / Chapter: 2" from top edge of paper (1.0in margin + 1.0in v)
        pagebreak(weak: true)
        v(1.0in)

/// Exhibit E: Table of Contents
/// AU DMin Manual (p. 7 & 9): TOC must reflect only first three levels of subheadings (depth: 4).
#let dmin-toc(title: auto, show-page-number: true, depth: 4) = {
  seen-chapter.update(false)
  seen-appendix.update(false)
  is-first-subhead.update(true)
  is-first-toc-entry.update(true)
  pagebreak(weak: true)
  if not show-page-number {
    set page(footer: none)
    v(1.0in)
    block(width: 100%, below: word-space-skip-2)[
      #align(center)[
        #set text(size: 12pt, weight: "regular")
        #set par(leading: word-single-leading, first-line-indent: 0pt)
        #context {
          let s = dmin-get-strings(au-dmin-lang.get())
          if title == auto { s.toc } else { upper(title) }
        }
      ]
    ]
    outline(title: none, depth: depth)
    pagebreak()
    set page(
      numbering: "i",
      footer: context {
        align(center)[
          #text(size: 12pt)[
            #counter(page).display("i")
          ]
        ]
      },
    )
  } else {
    set page(
      numbering: "i",
      footer: context {
        align(center)[
          #text(size: 12pt)[
            #counter(page).display("i")
          ]
        ]
      },
    )
    v(1.0in)
    block(width: 100%, below: word-space-skip-2)[
      #align(center)[
        #set text(size: 12pt, weight: "regular")
        #set par(leading: word-single-leading, first-line-indent: 0pt)
        #context {
          let s = dmin-get-strings(au-dmin-lang.get())
          if title == auto { s.toc } else { upper(title) }
        }
      ]
    ]
    outline(title: none, depth: depth)
  }
}
